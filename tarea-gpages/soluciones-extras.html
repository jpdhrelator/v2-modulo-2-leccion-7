Misión 11

Hobbies con jQuery dinámico

Mostrar un hobby al azar al cargar, cambiar entre ellos con fade y revelar horas semanales al doble clic.

HTML (dentro de la carta)

<div class="seccion" id="hobby-box">
  <strong>Hobby al azar</strong>
  <span id="hobby-nombre"></span> — <span id="hobby-desc"></span>
  <span id="hobby-horas"></span>
</div>
<button id="btn-hobby">Otro hobby</button>

jQuery

$(document).ready(function () {

  // Array de hobbies del alumno
  const hobbies = [
    { nombre: "Cocinar", descripcion: "Pasta los domingos", horas: 4 },
    { nombre: "Gaming",  descripcion: "Indie y estrategia",  horas: 6 },
    { nombre: "Leer",    descripcion: "Ciencia ficción",     horas: 3 },
    { nombre: "Dibujar", descripcion: "Sketches a lápiz",    horas: 2 }
  ];
  let i = Math.floor(Math.random() * hobbies.length);

  function pintarHobby() {
    $("#hobby-nombre").text(hobbies[i].nombre);
    $("#hobby-desc").text(hobbies[i].descripcion);
    $("#hobby-horas").text("");
  }
  pintarHobby();

  // Cambiar de hobby con fade
  $("#btn-hobby").click(function () {
    $("#hobby-box").fadeOut(200, function () {
      i = (i + 1) % hobbies.length;
      pintarHobby();
      $(this).fadeIn(200);
    });
  });

  // Doble clic = mostrar horas
  $("#hobby-box").dblclick(function () {
    $("#hobby-horas").text("~" + hobbies[i].horas + " hrs/semana");
  });

});
Misión 12

Modo Nocturno

Alterna una clase en el body para cambiar todo el tema de la página y actualiza el texto del botón.

HTML

<button id="btn-modo">Modo noche</button>

CSS

body {
  background: #fdf6ec;
  color: #2b2d42;
  transition: background 0.4s ease, color 0.4s ease;
}
body.modo-noche {
  background: #1a1a2e;
  color: #f4f1ec;
}
body.modo-noche .carta { box-shadow: 0 0 25px rgba(244, 196, 48, 0.3); }

jQuery

$(document).ready(function () {

  $("#btn-modo").click(function () {
    $("body").toggleClass("modo-noche");

    if ($("body").hasClass("modo-noche")) {
      $(this).text("Modo día");
    } else {
      $(this).text("Modo noche");
    }
  });

});
Misión 13

Reverso de la carta

Sección oculta con datos secretos que se revela con slideToggle y un botón que cambia su texto según el estado.

HTML

<div class="seccion" id="reverso">
  <strong>Reverso secreto</strong>
  Frase: "Compila o muere intentándolo."<br>
  Comida favorita: Sushi.<br>
  Canción para programar: Lofi beats infinitos.
</div>
<button id="btn-reverso">Ver reverso</button>

CSS

#reverso { display: none; }

jQuery

$(document).ready(function () {

  $("#btn-reverso").click(function () {
    $("#reverso").slideToggle(400);

    if ($(this).text() === "Ver reverso") {
      $(this).text("Ocultar reverso");
    } else {
      $(this).text("Ver reverso");
    }
  });

});
Misión 14

Ruleta de poderes secretos

Botón que elige un poder al azar, lo muestra con fade y hace brillar la carta brevemente.

HTML

<div class="seccion">
  <strong>Poder activo</strong>
  <div id="poder">—</div>
</div>
<button id="btn-poder">Lanzar poder</button>

CSS

.carta {
  transition: transform 0.3s ease, box-shadow 0.3s ease;
}
.carta.brillo {
  box-shadow: 0 0 35px gold;
  transform: scale(1.03);
}

jQuery

$(document).ready(function () {

  const poderes = [
    "Inmunidad al lunes",
    "Doble café gratis",
    "Visión de bug instantánea",
    "Compilar sin errores al primer intento",
    "Wi-Fi infinito",
    "Stack Overflow telepático"
  ];
  let ultimo = -1;

  $("#btn-poder").click(function () {

    // Sortear sin repetir el anterior
    let i = Math.floor(Math.random() * poderes.length);
    while (i === ultimo) {
      i = Math.floor(Math.random() * poderes.length);
    }
    ultimo = i;

    // Cambiar texto con fade
    $("#poder").fadeOut(200, function () {
      $(this).text(poderes[i]).fadeIn(200);
    });

    // Brillo temporal en la carta
    $(".carta").addClass("brillo").delay(500).queue(function (next) {
      $(this).removeClass("brillo");
      next();
    });
  });

});
